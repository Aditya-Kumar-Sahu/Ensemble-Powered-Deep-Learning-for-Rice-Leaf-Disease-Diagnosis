================================================================================
USAGE EXAMPLES
================================================================================

TRAINING
--------
python scripts/train.py \
    --data-dir /path/to/dataset \
    --model resnet50 \
    --epochs 15 \
    --batch-size 32

EVALUATION
----------
python scripts/evaluate.py \
    --data-dir /path/to/dataset \
    --ensemble \
    --output-dir results

INFERENCE
---------
python scripts/inference.py \
    --image leaf.jpg \
    --model resnet50 \
    --classes BrownSpot BacterialLeafBlight LeafSmut Healthy

WEB APPLICATION
---------------
python apps/gradio_app.py
# Access at http://localhost:7860

REST API
--------
python apps/fastapi_app.py
# API at http://localhost:8000
# Docs at http://localhost:8000/docs

DOCKER DEPLOYMENT
-----------------
docker-compose up -d
# API: http://localhost:8000
# Web: http://localhost:7860
